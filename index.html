function showGameOver(secs,kld){

  var isNew=scores.length===0||secs>scores[0].secs;

  var ov=document.getElementById('overlay');

  ov.innerHTML='<h1>💀 CHÂTEAU DÉTRUIT !</h1>'+

    '<div id="score-entry">'+

      '<div class="result">'+(isNew?'🎉 Nouveau record ! ':'')+' ⏱️ '+secs+'s — 👾 '+kld+' éliminé'+(kld>1?'s':'')+'</div>'+

      '<p>Entre ton prénom :</p>'+

      '<div id="name-row"><input id="name-input" type="text" maxlength="12" placeholder="Ton prénom..."/><button id="save-btn" onclick="saveScore('+secs+','+kld+')">✔ OK</button></div>'+

    '</div>'+

    '<div class="panel"><h2>📚 À RÉVISER</h2><div id="mistakes-inner">'+renderMistakes()+'</div></div>'+

    '<div class="panel"><h2>🏆 MEILLEURS SCORES</h2><div id="lb-inner">'+renderLB()+'</div></div>'+

    '<button class="bigbtn" onclick="startGame()">↺ Rejouer</button>';

  ov.style.display='flex';

  setTimeout(function(){

    var inp=document.getElementById('name-input');

    if(inp){inp.focus();inp.addEventListener('keydown',function(e){if(e.key==='Enter')saveScore(secs,kld);});}

  },100);

}

function saveScore(secs,kld){

  var inp=document.getElementById('name-input');

  var name=(inp&&inp.value.trim())?inp.value.trim():'Anonyme';

  addScore(name,secs,kld);updateBestHUD();

  var entry=document.getElementById('score-entry');

  if(entry)entry.innerHTML='<div class="result" style="color:#4ec94e">✅ Sauvegardé pour <b>'+name+'</b> !</div>';

  var lb=document.getElementById('lb-inner');if(lb)lb.innerHTML=renderLB();

}


function startGame(){

  if(AC.state==='suspended')AC.resume();

  document.getElementById('overlay').style.display='none';

  enemies=[];particles=[];shakeT=0;

  lives=4;killed=0;survivalSecs=0;errorMap={};

  gameState='playing';locked=false;

  updateHUD();

  document.getElementById('timer').textContent='0s';

  clearTimeout(spawnTO);clearInterval(timerInt);cancelAnimationFrame(raf);

  spawnEnemy(300);scheduleSpawn();

  timerInt=setInterval(function(){if(gameState!=='playing')return;survivalSecs++;document.getElementById('timer').textContent=survivalSecs+'s';},1000);

  render();showQ();

}

function endGame(){

  gameState='over';clearTimeout(spawnTO);clearInterval(timerInt);cancelAnimationFrame(raf);

  sfxDie();showGameOver(survivalSecs,killed);

}
